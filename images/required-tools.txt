# Binaries the cnmsql instance manager executes in the Percona Server image.
# images/check-tools.sh runs every built image against this list and fails the
# build when a binary is missing or its probe does not exit 0, so a slimming
# pass in a Dockerfile can't quietly remove a tool the operator needs.
#
# Format: <binary> [probe args]
#   probe args  arguments that make the binary print something and exit 0
#               (default: --version). "-" only checks that the binary is on
#               PATH and executable, for tools with no side-effect-free probe.
#
# Keep in sync with pkg/engine in cnmsql/cnmsql.

# Server; also initialises the data dir (mysqld --initialize-insecure).
mysqld
# Physical backups, clones and restores.
xtrabackup
xbstream
# Binlog archiving and PITR decode.
mysqlbinlog
# PITR replay, and loading logical dumps.
mysql
# Logical backups.
mysqldump
